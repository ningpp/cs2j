**Tech Stack:** Java 25, Maven, JUnit 5, C# Roslyn converter, xUnit.

```java
package io.github.ningpp.compat;

import java.util.Arrays;

/**
 * Lightweight array-slice container (generated by CSharpToJava converter).
 * Does not simulate C# ref-struct lifetime semantics.
 */
public final class Span<T> {
    private final T[] array;
    private final int offset;
    private final int length;

    public Span(T[] array) {
        this.array = array;
        this.offset = 0;
        this.length = array.length;
    }

    public Span(T[] array, int offset, int length) {
        if (offset < 0 || length < 0 || offset + length > array.length) {
            throw new IndexOutOfBoundsException(
                "offset=" + offset + ", length=" + length + ", arrayLength=" + array.length);
        }
        this.array = array;
        this.offset = offset;
        this.length = length;
    }

    public int length() {
        return length;
    }

    public T get(int index) {
        if (index < 0 || index >= length) {
            throw new IndexOutOfBoundsException("index=" + index + ", length=" + length);
        }
        return array[offset + index];
    }

    public void set(int index, T value) {
        if (index < 0 || index >= length) {
            throw new IndexOutOfBoundsException("index=" + index + ", length=" + length);
        }
        array[offset + index] = value;
    }

    public Span<T> slice(int start) {
        return slice(start, length - start);
    }

    public Span<T> slice(int start, int len) {
        if (start < 0 || len < 0 || start + len > length) {
            throw new IndexOutOfBoundsException(
                "start=" + start + ", len=" + len + ", length=" + length);
        }
        return new Span<>(array, offset + start, len);
    }

    public T[] toArray() {
        return Arrays.copyOfRange(array, offset, offset + length);
    }
}
```

```java
    // ---- TryFormat helpers ----

    private static boolean writeFormattedToSpan(String formatted, Span<Character> dest, IntHolder charsWritten) {
        if (formatted.length() > dest.length()) {
            return false;
        }
        for (int i = 0; i < formatted.length(); i++) {
            dest.set(i, formatted.charAt(i));
        }
        charsWritten.value = formatted.length();
        return true;
    }

    private static String formatValue(String format, Object value) {
        if (format == null || format.isEmpty()) {
            return String.valueOf(value);
        }
        return String.format(Locale.ROOT, "%" + format, value);
    }

    public static boolean tryFormatByte(byte value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value & 0xFF), dest, charsWritten);
    }

    public static boolean tryFormatSByte(byte value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, (int) value), dest, charsWritten);
    }

    public static boolean tryFormatShort(short value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, (int) value), dest, charsWritten);
    }

    public static boolean tryFormatUShort(short value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value & 0xFFFF), dest, charsWritten);
    }

    public static boolean tryFormatInt(int value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value), dest, charsWritten);
    }

    public static boolean tryFormatUInt(long value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, Long.toUnsignedString(value)), dest, charsWritten);
    }

    public static boolean tryFormatLong(long value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value), dest, charsWritten);
    }

    public static boolean tryFormatULong(long value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, toUnsignedString(value)), dest, charsWritten);
    }

    public static boolean tryFormatFloat(float value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value), dest, charsWritten);
    }

    public static boolean tryFormatDouble(double value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value), dest, charsWritten);
    }

    public static boolean tryFormatDecimal(Decimal value, Span<Character> dest, IntHolder charsWritten, String format) {
        String s;
        if (format == null || format.isEmpty()) {
            s = value.toString();
        } else {
            s = value.toString(format);
        }
        return writeFormattedToSpan(s, dest, charsWritten);
    }

    public static boolean tryFormatObject(Object value, Span<Character> dest, IntHolder charsWritten, String format) {
        return writeFormattedToSpan(formatValue(format, value), dest, charsWritten);
    }

    private static String toUnsignedString(long value) {
        if (value >= 0) {
            return Long.toString(value);
        }
        // Unsigned conversion for negative long values
        BigInteger upper = BigInteger.valueOf((value >>> 32) & 0xFFFFFFFFL);
        BigInteger lower = BigInteger.valueOf(value & 0xFFFFFFFFL);
        return upper.shiftLeft(32).add(lower).toString();
    }
```

```csharp
    private static string? GetTryFormatHelperMethod(ExpressionSyntax receiverExpression, ConversionContext context)
    {
        // TryFormat is an instance method on numeric primitives.
        // Check the receiver's type via semantic model.
        if (context.SemanticModel != null)
        {
            var receiverType = context.GetTypeInfo(receiverExpression).Type;
            if (receiverType != null)
            {
                return receiverType.SpecialType switch
                {
                    SpecialType.System_Byte    => "MathHelper.tryFormatByte",
                    SpecialType.System_SByte    => "MathHelper.tryFormatSByte",
                    SpecialType.System_Int16    => "MathHelper.tryFormatShort",
                    SpecialType.System_UInt16   => "MathHelper.tryFormatUShort",
                    SpecialType.System_Int32    => "MathHelper.tryFormatInt",
                    SpecialType.System_UInt32   => "MathHelper.tryFormatUInt",
                    SpecialType.System_Int64    => "MathHelper.tryFormatLong",
                    SpecialType.System_UInt64   => "MathHelper.tryFormatULong",
                    SpecialType.System_Single   => "MathHelper.tryFormatFloat",
                    SpecialType.System_Double   => "MathHelper.tryFormatDouble",
                    SpecialType.System_Decimal  => "MathHelper.tryFormatDecimal",
                    _ => null
                };
            }
        }

        return null;
    }
```

```csharp
        // Numeric TryFormat: a.TryFormat(buf, out w, "X2", null) → MathHelper.tryFormatXxx(a, buf, _wH, "X2")
        if (originalMethodName == "TryFormat" && node.ArgumentList.Arguments.Count >= 2)
        {
            var tryFormatHelper = GetTryFormatHelperMethod(memberAccess.Expression, context);
            if (tryFormatHelper != null)
            {
                // Determine how many arguments to include (skip trailing IFormatProvider)
                int argCount = node.ArgumentList.Arguments.Count;
                // The last argument may be IFormatProvider — skip it
                int maxArgs = argCount;
                if (argCount >= 4)
                {
                    // TryFormat(Span, out int, ReadOnlySpan<char>, IFormatProvider?)
                    // The 4th arg (index 3) is IFormatProvider — skip it
                    maxArgs = 3;
                }
                else if (argCount == 3)
                {
                    // TryFormat(Span, out int, ReadOnlySpan<char>) or TryFormat(Span, out int, IFormatProvider?)
                    // Check if 3rd arg is IFormatProvider by looking at the method symbol
                    if (earlyMethodSymbol != null && earlyMethodSymbol.Parameters.Length >= 3
                        && earlyMethodSymbol.Parameters[2].Type.ToDisplayString() == "System.IFormatProvider")
                    {
                        maxArgs = 2;
                    }
                }

                var tryFormatArgs = ArgumentTransformer.TransformArgumentList(
                    node.ArgumentList, context, facade, 0, methodSymbol, maxArgCount: maxArgs);

                // Prepend the receiver (the value being formatted) as the first argument
                var tryFormatReceiver = facade.Transform(memberAccess.Expression, context);
                tryFormatArgs = string.IsNullOrEmpty(tryFormatArgs)
                    ? tryFormatReceiver
                    : $"{tryFormatReceiver}, {tryFormatArgs}";

                return $"{tryFormatHelper}({tryFormatArgs})";
            }
        }
```
